function doraMissingDataSummary(metrics) { const contracts = (metrics.missingData || []) .map((item) => item.platformContract || item.data || '') .filter(Boolean) .slice(0, 5); return contracts.length ? 'Missing production data: ' + contracts.join('; ') + '. Without these records the dashboard will keep showing estimates rather than true DORA performance.' : 'No missing DORA data is currently listed.'; }
